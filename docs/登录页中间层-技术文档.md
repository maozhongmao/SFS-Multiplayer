# SFS 1.6 PC：主菜单按钮与登录页中间层

页面链路：

```text
主菜单按钮
 -> MyLoginMenu.OpenMenu()
 -> 输入 Server / Username / Password
 -> Join()
 -> 校验参数
 -> 你自己的连接逻辑，或 WorldEntry.Open()
```

登录页继承 BasicMenu，用 Builder/MenuGenerator 创建，不要手写 Canvas。

## MyLoginMenu.cs

```csharp
using System;
using System.Net;
using SFS.UI;
using SFS.UI.ModGUI;
using TMPro;
using UnityEngine;

namespace MyMod
{
    public class MyLoginMenu : BasicMenu
    {
        public static MyLoginMenu main;
        public static Window window;
        public static GameObject windowHolder;
        private static readonly int windowID = Builder.GetRandomID();

        private static readonly Vector2Int windowSize = new Vector2Int(760, 520);
        private const int ContentWidth = 700;
        private const int RowGap = 16;
        private const int LabelWidth = 180;
        private const int InputWidth = 480;
        private const int RowHeight = 50;
        private const int ButtonWidth = 300;
        private const int ButtonHeight = 80;

        private TextInput input_server;
        private TextInput input_username;
        private TextInput input_password;

        protected override CloseMode OnEscape => CloseMode.Current;

        public static void OpenMenu()
        {
            windowHolder = Builder.CreateHolder(
                Builder.SceneToAttach.CurrentScene,
                "MyMod - Login Menu Holder");

            main = windowHolder.AddComponent<MyLoginMenu>();
            main.OnOpen();
        }

        public override void OnOpen()
        {
            if (ScreenManager.main.CurrentScreen != this)
            {
                ScreenManager.main.OpenScreen(() => this);
                windowHolder.SetActive(true);

                window = Builder.CreateWindow(
                    windowHolder.transform,
                    windowID,
                    windowSize.x,
                    windowSize.y,
                    0,
                    windowSize.y / 2,
                    draggable: false,
                    savePosition: false,
                    1f,
                    "My Mod - Login");

                CreateUI();
            }
        }

        public override void Close()
        {
            if (ScreenManager.main.CurrentScreen == this && windowHolder != null)
            {
                ScreenManager.main.CloseCurrent();
                windowHolder.SetActive(false);
            }
        }

        private void CreateUI()
        {
            window.CreateLayoutGroup(
                SFS.UI.ModGUI.Type.Vertical,
                childAlignment: TextAnchor.MiddleCenter,
                spacing: RowGap,
                padding: new RectOffset(20, 20, 20, 20),
                disableChildSizeControl: true);

            Container row = Builder.CreateContainer(window);
            row.CreateLayoutGroup(
                SFS.UI.ModGUI.Type.Horizontal,
                childAlignment: TextAnchor.MiddleCenter,
                spacing: RowGap,
                disableChildSizeControl: true);

            Container labels = Builder.CreateContainer(row);
            Container inputs = Builder.CreateContainer(row);
            labels.CreateLayoutGroup(SFS.UI.ModGUI.Type.Vertical, TextAnchor.MiddleLeft, spacing: RowGap, disableChildSizeControl: true);
            inputs.CreateLayoutGroup(SFS.UI.ModGUI.Type.Vertical, TextAnchor.MiddleLeft, spacing: RowGap, disableChildSizeControl: true);

            Builder.CreateLabel(labels, LabelWidth, RowHeight, 0, 0, "Server").TextAlignment = TextAlignmentOptions.MidlineLeft;
            input_server = Builder.CreateTextInput(
                inputs, InputWidth, RowHeight, 0, 0,
                "127.0.0.1:9806",
                delegate(string value) { });

            Builder.CreateLabel(labels, LabelWidth, RowHeight, 0, 0, "Username").TextAlignment = TextAlignmentOptions.MidlineLeft;
            input_username = Builder.CreateTextInput(
                inputs, InputWidth, RowHeight, 0, 0,
                "player",
                delegate(string value) { });

            Builder.CreateLabel(labels, LabelWidth, RowHeight, 0, 0, "Password").TextAlignment = TextAlignmentOptions.MidlineLeft;
            input_password = Builder.CreateTextInput(
                inputs, InputWidth, RowHeight, 0, 0,
                "",
                delegate(string value) { });

            Container buttons = Builder.CreateContainer(window);
            buttons.CreateLayoutGroup(
                SFS.UI.ModGUI.Type.Horizontal,
                childAlignment: TextAnchor.MiddleCenter,
                spacing: RowGap,
                disableChildSizeControl: true);

            Builder.CreateButton(buttons, ButtonWidth, ButtonHeight, 0, 0, Close, "Back");
            Builder.CreateButton(buttons, ButtonWidth, ButtonHeight, 0, 0, Join, "Join");
        }

        private void Join()
        {
            string server = (input_server.Text ?? string.Empty).Trim();
            string username = (input_username.Text ?? string.Empty).Trim();
            string password = input_password.Text ?? string.Empty;

            if (server.Length == 0)
            {
                MsgDrawer.main.Log("Server cannot be empty");
                return;
            }

            if (username.Length == 0)
            {
                MsgDrawer.main.Log("Username cannot be empty");
                return;
            }

            // 简单解析 IPv4/域名:端口；生产代码要补 IPv6 方括号和 DNS。
            string host = server;
            int port = 9806;
            int sep = server.LastIndexOf(':');
            if (sep > 0)
            {
                host = server.Substring(0, sep);
                if (!int.TryParse(server.Substring(sep + 1), out port) || port < 1 || port > 65535)
                {
                    MsgDrawer.main.Log("Invalid port");
                    return;
                }
            }

            if (!IPAddress.TryParse(host, out IPAddress address))
            {
                MsgDrawer.main.Log("Only IPv4 is supported in this sample");
                return;
            }

            MsgDrawer.main.Log("Connecting to " + address + ":" + port + " ...");

            // 在这里接你自己的连接逻辑。
            // 如果要完全复用联机登录：
            // var info = new MultiplayerSFS.Mod.JoinInfo
            // {
            //     address = address,
            //     port = port,
            //     username = username,
            //     password = password
            // };
            // _ = MultiplayerSFS.Mod.ClientManager.TryConnect(info);

            // 如果这个登录页只是自定义单机/私服入口：
            WorldEntry.Open();
        }
    }
}
```

## 主菜单按钮怎么打开这个页面

把 MainMenuButtonMod 里的点击回调改成：

```csharp
button.clickEvent.AddListener(delegate
{
    SoundPlayer.main.clickSound.Play();
    MyLoginMenu.OpenMenu();
});
```

## 常见坑

1. OpenMenu 必须走 Builder.CreateHolder + AddComponent + OnOpen。
2. OnOpen 里必须判断 ScreenManager.main.CurrentScreen != this，避免重复建窗。
3. Close 必须同时 CloseCurrent 和隐藏 windowHolder。
4. TextInput.Text 要取当前值，不要依赖 onChange 回调；回调可能落后一次输入。
5. PC 端输入框只支持 ASCII，中文会显示方框；需要的话按游戏现有做法标红。
6. 登录页只负责收集参数；真正进世界仍由 WorldEntry 或你的网络层负责。
